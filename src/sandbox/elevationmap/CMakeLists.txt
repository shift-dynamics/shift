cmake_minimum_required(VERSION 2.8.11)
project(elevationmap_test)


find_package(ImageMagick REQUIRED)
find_package(ImageMagick COMPONENTS MagickWand REQUIRED)
add_executable(elevationmap main.c)
target_link_libraries(elevationmap ${ImageMagick_LIBRARIES} ${ImageMagick_MagickWand_LIBRARIES})
include_directories(${ImageMagick_INCLUDE_DIRS} ${ImageMagick_MagickWand_INCLUDE_DIRS})
add_definitions(-DMAGICKCORE_HDRI_ENABLE=1)
add_definitions(-DMAGICKCORE_QUANTUM_DEPTH=16)
add_definitions(-ggdb)
find_library( M_LIB m REQUIRED )
include_directories(../../ksl)
include_directories(../../core)
add_library(ksl STATIC ../../ksl/ksl.c)
add_definitions(-ggdb)
add_library(elevationmaplib STATIC ../../core/elevationmap.c)
add_definitions(-ggdb)
target_link_libraries(elevationmap ${M_LIB} elevationmaplib ksl)
